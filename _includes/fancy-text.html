{%- comment -%} Turns styling markers in a short line of copy into styles: ==words== highlighter behind the words (big titles) ~~words~~ hand-drawn underline under the words (small titles, descriptions) Usage: {% include fancy-text.html text=s.help.heading %} {%- endcomment -%} {%- assign parts = include.text | split: "==" -%} {%- capture marked -%} {%- for part in parts -%} {%- assign odd = forloop.index0 | modulo: 2 -%} {%- if odd == 1 -%}{{ part }}{%- else -%}{{ part }}{%- endif -%} {%- endfor -%} {%- endcapture -%} {%- assign parts = marked | split: "~~" -%} {%- for part in parts -%} {%- assign odd = forloop.index0 | modulo: 2 -%} {%- if odd == 1 -%}{{ part }}{%- else -%}{{ part }}{%- endif -%} {%- endfor -%}
